// Carrega e processa o EPG com suporte total a navegadores mobile (HTTPS Proxy) async function carregarEpgCompleto(usuario, senha) { const selectCanal = document.getElementById('selectCanalEpg'); selectCanal.innerHTML = ''; // Link original em HTTP const urlEpgOriginal = `http://plus.lat/xmltv.php?username=${usuario}&password=${senha}`; // Proxy HTTPS para liberar o acesso no celular const urlEpg = `https://corsproxy.io/?${encodeURIComponent(urlEpgOriginal)}`; try { const response = await fetch(urlEpg); if (!response.ok) throw new Error("Erro ao carregar dados do EPG"); const textoXml = await response.text(); const parser = new DOMParser(); const xmlDoc = parser.parseFromString(textoXml, "text/xml"); const programas = xmlDoc.getElementsByTagName("programme"); const canais = xmlDoc.getElementsByTagName("channel"); dadosEpgGlobais.canaisUnicos = {}; dadosEpgGlobais.programasPorCanal = {}; const mapaIdsParaNomeUnico = {}; // 1. Processa canais limpando resoluções (SD, HD, FHD) e agrupa por Nome Único for (let i = 0; i < canais.length; i++) { const idOriginal = canais[i].getAttribute("id"); const nomeBruto = canais[i].getElementsByTagName("display-name")[0]?.textContent || idOriginal; const nomeLimpo = normalizarNomeCanal(nomeBruto); if (nomeLimpo) { mapaIdsParaNomeUnico[idOriginal] = nomeLimpo; if (!dadosEpgGlobais.canaisUnicos[nomeLimpo]) { dadosEpgGlobais.canaisUnicos[nomeLimpo] = true; dadosEpgGlobais.programasPorCanal[nomeLimpo] = []; } } } const agora = new Date(); const meiaNoite = new Date(); meiaNoite.setHours(23, 59, 59, 999); // 2. Associa a programação ao canal limpo unificado for (let i = 0; i < programas.length; i++) { const prog = programas[i]; const channelIdOriginal = prog.getAttribute("channel"); const nomeLimpo = mapaIdsParaNomeUnico[channelIdOriginal]; if (nomeLimpo && dadosEpgGlobais.programasPorCanal[nomeLimpo]) { const startStr = prog.getAttribute("start"); const stopStr = prog.getAttribute("stop"); const titulo = prog.getElementsByTagName("title")[0]?.textContent || "Programa sem título"; const descricao = prog.getElementsByTagName("desc")[0]?.textContent || "Sem descrição disponível."; const dataInicio = parseXmlTvDate(startStr); const dataFim = parseXmlTvDate(stopStr); if (dataInicio && dataFim && dataFim >= agora && dataInicio <= meiaNoite) { const jaExiste = dadosEpgGlobais.programasPorCanal[nomeLimpo].some( p => p.titulo === titulo && p.inicio.getTime() === dataInicio.getTime() ); if (!jaExiste) { dadosEpgGlobais.programasPorCanal[nomeLimpo].push({ titulo, descricao, inicio: dataInicio, fim: dataFim, rolandoAgora: agora >= dataInicio && agora <= dataFim }); } } } } renderizarSeletorCanaisPorCategoria(); } catch (e) { console.error("Erro EPG:", e); selectCanal.innerHTML = ''; } }
